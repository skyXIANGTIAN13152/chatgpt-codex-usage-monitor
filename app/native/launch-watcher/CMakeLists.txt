cmake_minimum_required(VERSION 3.20)
project(tiga_launch_watcher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(watcher_core STATIC watcher_core.cpp ../core-helper/upstream/src/json.cpp)
target_include_directories(watcher_core PUBLIC . ../core-helper/upstream/include)
target_compile_definitions(watcher_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(watcher_core PRIVATE /utf-8 /W4)
endif()
add_executable(tiga-launch-watcher WIN32 main.cpp)
target_link_libraries(tiga-launch-watcher PRIVATE watcher_core shell32)
set_target_properties(tiga-launch-watcher PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../runtime")
if(MSVC)
  target_compile_options(tiga-launch-watcher PRIVATE /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(launch_watcher_tests tests.cpp)
  target_link_libraries(launch_watcher_tests PRIVATE watcher_core)
  if(MSVC)
    target_compile_options(launch_watcher_tests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME launch_watcher_core COMMAND launch_watcher_tests)
endif()
